function forecastOrder(o,includeSaturday=false){
 const tasks=orderTasks(o.id).filter(t=>t.status!=='done');
 if(!tasks.length)return {finish:isoDate(new Date()),impossible:false};
 const today=isoDate(new Date());
 const used={}; // minutes of THIS order assigned by the forecast per date/employee
 const key=(d,e)=>d+'|'+e;
 function free(d,e,allowSaturday=includeSaturday){return Math.max(0,employeeCapacity(d,e,allowSaturday)-employeePlannedMinutes(d,e,o.id)-(used[key(d,e)]||0))}
 function alloc(t,earliest){
   let need=Math.max(0,Number(t.estimate)||0);if(need===0)return earliest;
   let start=earliest;
   if(t.date&&t.date>start)start=t.date;
   const fixed=t.employee||'';
   const explicitSaturday=!!(t.date&&parseDate(t.date).getDay()===6);
   let last=start,guard=0;
   while(need>0&&guard++<180){
     const candidates=fixed?[fixed]:EMPLOYEES.slice();
     let best=null,bestFree=0;
     const allowSat=includeSaturday||(explicitSaturday&&start===t.date);
     for(const e of candidates){const f=free(start,e,allowSat);if(f>bestFree){best=e;bestFree=f}}
     if(best&&bestFree>0){const take=Math.min(need,bestFree);used[key(start,best)]=(used[key(start,best)]||0)+take;need-=take;last=start;}
     if(need>0)start=addDays(start,1);
   }
   return last;
 }
 const finishes={};let orderFinish=today;
 for(const t of tasks){const prev=prevTask(t);let earliest=today;if(t.dependsPrev&&prev&&prev.status!=='done'){const pf=finishes[prev.id];if(pf)earliest=pf;else if(taskFinishDate(prev))earliest=taskFinishDate(prev);}let f;if(isDryTask(t)){const startAt=t.waitStartAt||dtString(earliest,'00:00');f=dtDate(t.waitEndAt||addMinutesDT(startAt,Number(t.estimate)||(CFG.rules?.dryRoomMinutes||540)));}else if(isExternalTask(t)){const lead=Number(t.externalLeadDays)||(CFG.rules?.externalLeadDays||14);const base=t.externalSentDate||t.date||earliest;f=t.expectedReturnDate||addDays(base,lead);}else{f=alloc(t,earliest)}finishes[t.id]=f;if(f>orderFinish)orderFinish=f;}
 return {finish:orderFinish,impossible:false};
}
