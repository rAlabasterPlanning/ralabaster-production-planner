function freeIntervalsForEmployee(date,emp,includeSaturday=false,excludeIds=new Set()){
 const cap=employeeCapacity(date,emp,includeSaturday);if(cap<=0)return[];
 const ws=timeToMin(dayStartTime(date,emp)),we=workdayEndMinutes(date,emp);if(we<=ws)return[];
 const busy=[];
 for(const x of scheduledEntries(date,emp)){
   if(excludeIds.has(x.task.id))continue;
   const st=timeToMin(x.seg.start||dayStartTime(date,emp));const en=Math.min(we,st+(Number(x.seg.minutes)||0));if(en>ws&&st<we)busy.push([Math.max(ws,st),en]);
 }
 busy.sort((a,b)=>a[0]-b[0]);const merged=[];for(const b of busy){if(!merged.length||b[0]>merged.at(-1)[1])merged.push(b);else merged.at(-1)[1]=Math.max(merged.at(-1)[1],b[1])}
 const gaps=[];let cur=ws;for(const [a,b] of merged){if(a>cur)gaps.push([cur,a]);cur=Math.max(cur,b)}if(cur<we)gaps.push([cur,we]);return gaps;
}
function scheduleTaskPrecisely(t,emp,startAt,includeSaturday=false,excludeIds=new Set()){
 clearTaskPlanning(t);let need=Math.max(0,Number(t.estimate)||0),date=dtDate(startAt),firstDate=date,earliest=timeToMin(dtTime(startAt)),segs=[],guard=0;
 while(need>0&&guard++<180){const gaps=freeIntervalsForEmployee(date,emp,includeSaturday,excludeIds);for(const [a,b] of gaps){let st=a;if(date===firstDate)st=Math.max(st,earliest);if(st>=b)continue;const take=Math.min(need,b-st);if(take>0){segs.push({date,employee:emp,start:minToTime(st),minutes:take});need-=take;if(need<=0)break}}if(need>0){date=addDays(date,1);earliest=0}}
 t.planSegments=segs;t.employee=emp;t.date=segs[0]?.date||null;t.start=segs[0]?.start||'';return taskFinishAt(t);
}
function dynamicReplanAfterCompletion(t,completedAt){
 const chain=dependentChainAfter(t);if(!chain.length)return;
 const preserved=new Map(chain.map(x=>[x.id,{employee:x.employee||firstSegment(x)?.employee||null,status:x.status,externalSentDate:x.externalSentDate||''}]));
 // Eerst vrijmaken, zodat oude blokken van dezelfde keten de nieuwe planning niet tegenhouden.
 for(const x of chain){if(isExternalTask(x)&&x.status==='external'&&x.externalSentDate)continue;clearTaskPlanning(x)}
 let cursor=completedAt;
 for(const x of chain){
   if(isDryTask(x)){scheduleDryTask(x,cursor);cursor=x.waitEndAt;continue}
   if(isExternalTask(x)){
     if(x.status==='external'&&x.externalSentDate){cursor=dtString(expectedExternalReturn(x),'00:00');continue}
     const d=dtDate(cursor);x.employee=null;x.planSegments=[];x.date=d;x.status='open';x.expectedReturnDate=addDays(d,Number(x.externalLeadDays)||(CFG.rules?.externalLeadDays||14));cursor=dtString(x.expectedReturnDate,'00:00');continue
   }
   const emp=preserved.get(x.id)?.employee;if(!emp){clearTaskPlanning(x);break}
   scheduleTaskPrecisely(x,emp,cursor,false,new Set(chain.map(y=>y.id)));cursor=taskFinishAt(x)||cursor;
 }
}
