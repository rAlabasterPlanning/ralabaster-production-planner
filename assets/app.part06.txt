function readRollChoices(){return [...document.querySelectorAll('#rollRows tr')].map(r=>({id:r.dataset.id,remaining:+r.dataset.rem,strategy:r.querySelector('.rollStrategy').value,date:r.querySelector('.rollDate').value,scope:r.querySelector('.rollScope').value}))}
function applyRollChoices(choices){for(const c of choices){const t=state.tasks.find(x=>x.id===c.id);if(!t)continue;const emp=t.employee||firstSegment(t)?.employee;if(!emp){clearTaskPlanning(t);continue}t.estimate=c.remaining;t.actual=0;t.doneQty=t.status==='partial'?t.doneQty||0:0;t.status='open';let startDate;if(c.strategy==='manual')startDate=c.date;else startDate=nextWorkingDateFor(emp,selectedDate,false,false);
  let displaced=[];
  if(c.strategy==='next_first'){
    // Maak echt ruimte aan het begin van de dag: bestaande taken van deze medewerker op die dag
    // worden opnieuw achter deze prioriteitstaak gepland en kunnen daardoor doorlopen naar een volgende werkdag.
    const seen=new Set();for(const x of scheduledEntries(startDate,emp)){if(x.task.id!==t.id&&!seen.has(x.task.id)){seen.add(x.task.id);displaced.push({task:x.task,old:firstSegment(x.task)});clearTaskPlanning(x.task)}}
  }
  scheduleTaskAcrossCapacity(t,emp,startDate,c.strategy==='next_first',false);
  if(displaced.length){displaced.sort((a,b)=>(a.old?.start||'99:99').localeCompare(b.old?.start||'99:99'));for(const d of displaced)scheduleTaskAcrossCapacity(d.task,emp,startDate,false,false)}
  const finish=taskFinishAt(t)||dtString(startDate,'00:00');if(c.scope==='chain')cascadeReplanOrder(t.orderId,t.seq,finish)}}
function processRollOpenTasks(){const choices=readRollChoices();const backup=JSON.parse(JSON.stringify(state));applyRollChoices(choices);const late=state.orders.filter(o=>o.active&&o.deadline).map(o=>({o,a:orderAnalysis(o)})).filter(x=>x.a.finish>x.o.deadline);state=backup;if(late.length){const msg=`Door deze verschuiving halen de volgende orders waarschijnlijk de deadline niet:

${late.map(x=>`${x.o.orderNo} – ${x.o.product}: verwacht ${fmtDate(x.a.finish)}, deadline ${fmtDate(x.o.deadline)}`).join('\n')}

Toch verwerken?`;if(!confirm(msg))return}applyRollChoices(choices);save();closeModal();renderToday()}
