async function initSupabase(){
 const c=onlineConfig();if(!c.url||!c.anonKey||!window.supabase){cloudStatus='offline';renderOnlineBadge();return}
 try{
  supabaseClient=window.supabase.createClient(c.url,c.anonKey,{auth:{persistSession:true,autoRefreshToken:true,detectSessionInUrl:true}});
  const {data}=await supabaseClient.auth.getSession();cloudUser=data?.session?.user||null;
  if(cloudUser){await loadCloudState();startCloudRefresh()}else{cloudStatus='login';renderOnlineBadge()}
  supabaseClient.auth.onAuthStateChange(async(_event,session)=>{cloudUser=session?.user||null;if(cloudUser){await loadCloudState();startCloudRefresh()}else{stopCloudRefresh();cloudStatus='login';renderOnlineBadge();renderSettings()}})
 }catch(e){console.error(e);cloudStatus='error';renderOnlineBadge()}
}

function isoDate(d){const z=new Date(d.getTime()-d.getTimezoneOffset()*60000);return z.toISOString().slice(0,10)}
function parseDate(s){const [y,m,d]=s.split('-').map(Number);return new Date(y,m-1,d)}
function addDays(s,n){const d=parseDate(s);d.setDate(d.getDate()+n);return isoDate(d)}
function mondayOf(d){const x=new Date(d);const day=(x.getDay()+6)%7;x.setDate(x.getDate()-day);x.setHours(12,0,0,0);return x}
function fmtDate(s){return parseDate(s).toLocaleDateString('nl-NL',{weekday:'long',day:'numeric',month:'long',year:'numeric'})}
function fmtShort(s){return parseDate(s).toLocaleDateString('nl-NL',{weekday:'short',day:'numeric',month:'numeric'})}
function fmtMin(m){m=Number(m)||0;return Math.floor(m/60)+'u '+String(m%60).padStart(2,'0')+'m'}
function esc(s){return String(s??'').replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]))}
function order(id){return state.orders.find(o=>o.id===id)}

function nowLocalDT(){const d=new Date(),z=new Date(d.getTime()-d.getTimezoneOffset()*60000);return z.toISOString().slice(0,16)}
function isTaskComplete(t){return t?.status==='done'||(isDryTask(t)&&t.waitEndAt&&t.waitEndAt<=nowLocalDT())}
function isBlocked(t){if(!t.dependsPrev||t.seq===1||isTaskComplete(t)) return false;const p=prevTask(t);return !!p && !isTaskComplete(p)}
function isExternalTask(t){return t?.type==='external'||/\bextern(?:e|al)?\b/i.test((t?.name||'')+' '+(t?.machine||''))}
function isDryTask(t){return t?.type==='wait'||/droogruimte/i.test((t?.name||'')+' '+(t?.machine||''))}

function displayStatus(t){if(isTaskComplete(t))return'done';if(t.status==='external')return'external';if(t.status==='partial')return'partial';if(t.status==='in_progress')return'in_progress';if(isBlocked(t))return'blocked';return hasPlanning(t)?'planned':'open'}
function statusLabel(t){const s=displayStatus(t);return {done:'✓ Gereed',external:'↗ Bij externe partij',partial:'◐ Deels gereed',in_progress:'▶ Bezig',blocked:'🔒 Geblokkeerd',planned:'○ Gepland',open:'○ Niet gepland'}[s]}
function endTime(start,mins){if(!start)return'';let [h,m]=start.split(':').map(Number);let x=h*60+m+(Number(mins)||0);return String(Math.floor(x/60)%24).padStart(2,'0')+':'+String(x%60).padStart(2,'0')}
function dtString(date,time='00:00'){return date+'T'+time}

function addMinutesDT(x,mins){const d=new Date(x);d.setMinutes(d.getMinutes()+(Number(mins)||0));const z=new Date(d.getTime()-d.getTimezoneOffset()*60000);return z.toISOString().slice(0,16)}
function taskSegments(t){if(Array.isArray(t.planSegments)&&t.planSegments.length)return t.planSegments;return t.date&&t.employee?[{date:t.date,employee:t.employee,start:t.start||'',minutes:Number(t.estimate)||0}]:[]}

function durationBetween(start,end){return Math.max(0,timeToMin(end)-timeToMin(start))}
const NORMAL_DAY_MIN=durationBetween(CFG.schedule?.monThu?.start||'08:15',CFG.schedule?.monThu?.end||'16:30');
const FRIDAY_RALPH_MIN=durationBetween(CFG.schedule?.friday?.Ralph?.start||'08:15',CFG.schedule?.friday?.Ralph?.end||'15:00');
const SATURDAY_OT_MIN=durationBetween(CFG.schedule?.saturdayOvertime?.start||'08:00',CFG.schedule?.saturdayOvertime?.end||'12:00');
function employeeCapacity(date,emp,includeSaturday=false){const dow=parseDate(date).getDay();if(dow>=1&&dow<=4)return NORMAL_DAY_MIN;if(dow===5)return emp==='Ralph'?FRIDAY_RALPH_MIN:0;if(dow===6)return includeSaturday?SATURDAY_OT_MIN:0;return 0}
function workWindow(date,emp){const dow=parseDate(date).getDay();if(dow>=1&&dow<=4)return `${CFG.schedule?.monThu?.start||'08:15'}–${CFG.schedule?.monThu?.end||'16:30'}`;if(dow===5&&emp==='Ralph')return `${CFG.schedule?.friday?.Ralph?.start||'08:15'}–${CFG.schedule?.friday?.Ralph?.end||'15:00'}`;if(dow===6)return `${CFG.schedule?.saturdayOvertime?.start||'08:00'}–${CFG.schedule?.saturdayOvertime?.end||'12:00'} (overwerk)`;return 'Niet aanwezig'}
function nextRegularDateForEmployee(date,emp){let d=addDays(date,1),guard=0;while(guard++<30){if(employeeCapacity(d,emp||'Ralph',false)>0)return d;d=addDays(d,1)}return d}
function expectedExternalReturn(t){const sent=t.externalSentDate||t.date||'';const lead=Number(t.externalLeadDays)||(CFG.rules?.externalLeadDays||14);return t.expectedReturnDate||(sent?addDays(sent,lead):'')}

function workdaysBetween(a,b){if(!a||!b)return 999;let d=parseDate(a),end=parseDate(b),n=0,step=d<=end?1:-1;while((step>0&&d<=end)||(step<0&&d>=end)){if(d.getDay()>=1&&d.getDay()<=5)n+=step;d.setDate(d.getDate()+step)}return n}
function remainingMinutes(orderId){return orderTasks(orderId).filter(t=>t.status!=='done'&&!isExternalTask(t)&&!isDryTask(t)).reduce((a,t)=>a+(Number(t.estimate)||0),0)}
function unplannedMinutes(orderId){return orderTasks(orderId).filter(t=>t.status!=='done'&&!isExternalTask(t)&&!isDryTask(t)&&!hasPlanning(t)).reduce((a,t)=>a+(Number(t.estimate)||0),0)}
function employeePlannedMinutes(date,emp,excludeOrderId='',excludeTaskId=''){let total=0;for(const t of state.tasks){if(t.status==='done'||isExternalTask(t)||isDryTask(t)||(excludeOrderId&&t.orderId===excludeOrderId)||(excludeTaskId&&t.id===excludeTaskId))continue;for(const seg of taskSegments(t))if(seg.date===date&&seg.employee===emp)total+=Number(seg.minutes)||0}return total}
function plannedMinutesOn(date){return EMPLOYEES.reduce((a,e)=>a+employeePlannedMinutes(date,e),0)}
function employeeFreeMinutes(date,emp,includeSaturday=false,excludeOrderId='',excludeTaskId=''){return Math.max(0,employeeCapacity(date,emp,includeSaturday)-employeePlannedMinutes(date,emp,excludeOrderId,excludeTaskId))}
function teamCapacityOn(date,includeSaturday=false){return EMPLOYEES.reduce((a,e)=>a+employeeCapacity(date,e,includeSaturday),0)}
function teamFreeMinutesUntil(deadline,includeSaturday=false,excludeOrderId=''){if(!deadline)return 999999;let d=isoDate(new Date()),total=0,guard=0;while(d<=deadline&&guard++<120){total+=EMPLOYEES.reduce((a,e)=>a+employeeFreeMinutes(d,e,includeSaturday,excludeOrderId),0);d=addDays(d,1)}return total}
