function parseDuration(s){
 s=mdCellText(s).toLowerCase();let m=0,found=false;
 const h=s.match(/(\d+(?:[.,]\d+)?)\s*u(?:ur)?\b/i);const mm=s.match(/(\d+(?:[.,]\d+)?)\s*m(?:in)?\b/i);const sec=s.match(/(\d+(?:[.,]\d+)?)\s*s(?:ec)?\b/i);
 if(h){m+=parseFloat(h[1].replace(',','.'))*60;found=true}if(mm){m+=parseFloat(mm[1].replace(',','.'));found=true}if(sec){m+=parseFloat(sec[1].replace(',','.'))/60;found=true}
 return found?Math.round(m):0
}
function mdCellText(s){return (s||'').replace(/\*\*/g,'').replace(/\[([^\]]+)\]\([^)]*\)/g,'$1').replace(/<[^>]*>/g,'').replace(/&nbsp;|\u00a0/g,' ').trim()}
function classifyOdooCell(raw){
 const s=mdCellText(raw).replace(/\s+/g,' ').trim();
 if(!s)return {type:'empty',value:null,text:s};
 if(/€\s*$/.test(s)||/€/.test(s))return {type:'money',value:null,text:s};
 if(/\d\s*(?:u(?:ur)?|m(?:in)?|s(?:ec)?)\b/i.test(s))return {type:'time',value:parseDuration(s),text:s};
 const compact=s.replace(/\s/g,'');
 if(/^[-+]?\d{1,3}(?:\.\d{3})*(?:,\d+)?$/.test(compact)||/^[-+]?\d+(?:[.,]\d+)?$/.test(compact)){
   let n=compact;
   if(n.includes(',')&&n.includes('.'))n=n.replace(/\./g,'').replace(',','.');
   else if(n.includes(','))n=n.replace(',','.');
   return {type:'number',value:parseFloat(n),text:s};
 }
 return {type:'text',value:null,text:s}
}
function splitOperationMachine(label){
 label=mdCellText(label).replace(/\s+/g,' ').trim();
 if(!label)return {name:'',machine:''};
 const parts=label.split(/\s+-\s+/).map(x=>x.trim()).filter(Boolean);
 if(parts.length===1)return {name:label,machine:label};
 if(parts.length%2===0){const half=parts.length/2;const a=parts.slice(0,half).join(' - '),b=parts.slice(half).join(' - ');if(a.toLowerCase()===b.toLowerCase())return {name:a,machine:b}}
 for(let i=1;i<parts.length;i++){const a=parts.slice(0,i).join(' - '),b=parts.slice(i).join(' - ');if(a.toLowerCase()===b.toLowerCase())return {name:a,machine:b}}
 return {name:parts[0],machine:parts.slice(1).join(' - ')}
}
function parseOdooRows(text){
 const lines=text.split(/\r?\n/).map(x=>x.trim()).filter(x=>x.length);
 const rows=[];
 for(const line of lines){
   if(/^\|/.test(line)){
     const cells=line.replace(/^\|/,'').replace(/\|$/,'').split('|').map(x=>x.trim());
     if(cells.length>=2)rows.push(cells);
   }else if(line.includes('\t')){
     const cells=line.split(/\t+/).map(x=>x.trim());if(cells.length>=2)rows.push(cells);
   }
 }
 return rows
}
function externalNameFromMarker(label){
 let x=mdCellText(label).replace(/^Te\s+/i,'').replace(/^Klaar\s+met\s+/i,'').trim();
 if(!x)return '';
 // Houd de Odoo-benaming herkenbaar, maar maak er één duidelijke externe taak van.
 return x.replace(/^kosten\s+/i,'').trim()
}
function parseImportText(){
 const text=document.getElementById('impText').value.trim();if(!text)return alert('Plak eerst Odoo-data of een bewerkingslijst.');

 const rows=parseOdooRows(text);
 if(rows.length>=2){
   let operationsStarted=false,productFound=false;
   for(const raw of rows){
     const c0=mdCellText(raw[0]||''), c1=classifyOdooCell(raw[1]||''), c2=classifyOdooCell(raw[2]||'');
     if(/^[-: ]+$/.test(c0)||/product.*aantal.*totale kosten/i.test(c0)||/kostprijs per eenheid/i.test(c0))continue;

     // Een lege omschrijving + tijd is in Odoo de totaalregel die materiaal/subcontracting van echte bewerkingen scheidt.
     if(!c0 && c1.type==='time'){operationsStarted=true;continue}
     if(!c0)continue;

     // Eerste inhoudelijke regel met een kaal getal in kolom 2 en geld in kolom 3 = hoofdproduct + aantal.
     if(!productFound && c1.type==='number' && (c2.type==='money'||raw.length>=2)){
       detectedProduct=c0;detectedQty=c1.value;productFound=true;continue
     }

     if(!operationsStarted){
       // Regels vóór totaaluren zijn materiaal/kosten/subcontracting-indicatoren, geen interne taken.
       if(/^Kosten\b/i.test(c0))continue;
       const te=c0.match(/^Te\s+(.+)$/i),klaar=c0.match(/^Klaar\s+met\s+(.+)$/i);
       if(te)externalMarkers.push(externalNameFromMarker(te[1]));
       if(klaar)externalMarkers.push(externalNameFromMarker(klaar[1]));
       continue
     }

     // Na de totaalregel: alleen een tijd in kolom 2 maakt een echte bewerking. Geld/aantallen worden nooit als tijd geïnterpreteerd.
     if(c1.type==='time' && c1.value>0){
       const om=splitOperationMachine(c0);const dry=/droogruimte/i.test(c0);const ext=/\bextern\b/i.test(c0);
       parsed.push({...om,estimate:dry?540:c1.value,type:dry?'wait':(ext?'external':'work')});
     }
   }

   // Combineer Te X + Klaar met X tot precies één externe stap van standaard 14 kalenderdagen.
   const key=x=>x.toLowerCase().replace(/\s+/g,' ').replace(/ten$/,'t').replace(/en$/,'').trim();
   const uniq=[];for(const m of externalMarkers){if(m&&!uniq.some(x=>key(x)===key(m)))uniq.push(m)}

     const label=m.trim();const display=/waterjet/i.test(label)?'Waterjetten':label;
     const task={name:`${display} extern`,machine:`${display} extern`,estimate:0,type:'external',dependsPrev:true};
     const idx=parsed.findIndex(x=>/alabaster klaarzetten/i.test(x.name+' '+x.machine));parsed.splice(idx>=0?idx+1:0,0,task)
   }
 }else{
   // Fallback voor plakken uit Odoo waarbij omschrijving en tijd op aparte regels terechtkomen.
   const lines=text.split(/\r?\n/).map(x=>mdCellText(x).replace(/\s+/g,' ').trim()).filter(Boolean);
   let pendingLabel='', seenTotal=false, productSet=false;
   for(const line of lines){
     if(/^[-|: ]+$/.test(line)||/product.*aantal.*totale kosten/i.test(line)||/kostprijs per eenheid/i.test(line))continue;
     const dur=parseDuration(line);const hasMoney=/€/.test(line);
     if(dur){
       if(!pendingLabel){seenTotal=true;continue}
       if(seenTotal||parsed.length){const om=splitOperationMachine(pendingLabel);if(om.name){const dry=/droogruimte/i.test(pendingLabel),ext=/\bextern\b/i.test(pendingLabel);parsed.push({...om,estimate:dry?540:dur,type:dry?'wait':(ext?'external':'work')})}}
       pendingLabel='';continue
     }
     if(hasMoney)continue;
     const num=line.replace(/\s/g,'');
     if(!productSet&&/^\d+(?:[.,]\d+)?$/.test(num)){detectedQty=parseFloat(num.replace(',','.'));productSet=true;continue}
     if(/^Kosten\b/i.test(line))continue;
     const te=line.match(/^Te\s+(.+)$/i),klaar=line.match(/^Klaar\s+met\s+(.+)$/i);
     if(te){externalMarkers.push(externalNameFromMarker(te[1]));continue}if(klaar){externalMarkers.push(externalNameFromMarker(klaar[1]));continue}
     if(!detectedProduct&&!/^\d/.test(line)){detectedProduct=line;continue}
     pendingLabel=line;
   }
   const key=x=>x.toLowerCase().replace(/\s+/g,' ').replace(/ten$/,'t').replace(/en$/,'').trim();const uniq=[];for(const m of externalMarkers){if(m&&!uniq.some(x=>key(x)===key(m)))uniq.push(m)};for(const m of uniq){const display=/waterjet/i.test(m)?'Waterjetten':m;const task={name:`${display} extern`,machine:`${display} extern`,estimate:0,type:'external',dependsPrev:true};const idx=parsed.findIndex(x=>/alabaster klaarzetten/i.test(x.name+' '+x.machine));parsed.splice(idx>=0?idx+1:0,0,task)}
 }
 if(detectedProduct&&!document.getElementById('impProduct').value.trim())document.getElementById('impProduct').value=detectedProduct;
 if(detectedQty&&(+document.getElementById('impQty').value===1||!document.getElementById('impQty').value))document.getElementById('impQty').value=detectedQty;

}
